package mysql

import (
	"github.com/jonbaldie/database/internal/catalog"
	"testing"
)

func TestFormalAutoIncrement(t *testing.T) {
	table := catalog.Table{Name: "items", Columns: []string{"id"}, ColumnTypes: []string{"TINYINT UNSIGNED"}, ColumnAttributes: []catalog.ColumnAttribute{{AutoIncrement: true}}}
	state := autoIncrementState{next: 1}
	for value := uint64(1); value <= 255; value++ {
		row := []string{storedSQLNullValue}
		next, err := fillAutoIncrementValue(table, row, state, 1)
		if err != nil {
			t.Fatal(err)
		}
		parsed, ok := autoIncrementPositiveValue(row[0])
		if !ok || parsed != value {
			t.Fatal("allocation repeated or skipped an identifier", value, row)
		}
		if value == 255 && !next.exhausted {
			t.Fatal("maximum identifier did not exhaust allocation")
		}
		state = next
	}
	row := []string{storedSQLNullValue}
	if _, err := fillAutoIncrementValue(table, row, state, 1); err == nil {
		t.Fatal("exhausted allocator accepted another identifier")
	}
	if row[0] != storedSQLNullValue {
		t.Fatal("failed allocation changed the row")
	}
	for limit := uint64(1); limit <= 255; limit++ {
		for value := uint64(1); value <= limit; value++ {
			state := advanceAutoIncrement(autoIncrementState{next: 1}, value, limit)
			if value == limit {
				if !state.exhausted || state.next != limit {
					t.Fatal("explicit maximum did not exhaust allocation")
				}
			} else if state.exhausted || state.next != value+1 {
				t.Fatal("explicit identifier did not advance allocation", value, limit, state)
			}
			lower := advanceAutoIncrement(state, 1, limit)
			if lower != state {
				t.Fatal("lower explicit identifier moved allocation backward")
			}
		}
	}
}
