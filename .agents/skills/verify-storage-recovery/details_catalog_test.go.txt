package catalog

import (
	"errors"
	"github.com/jonbaldie/database/internal/storage"
	"testing"
)

func formalCatalogDefinition(rows [][]string) Definition {
	return Definition{Namespaces: map[string]Namespace{"app": {Name: "app", Tables: map[string]Table{"items": {Name: "items", Columns: []string{"id"}, Rows: rows}}}}}
}

func TestFormalCatalogHydration(t *testing.T) {
	for _, durable := range []bool{false, true} {
		t.Run(map[bool]string{false: "seed legacy", true: "prefer durable"}[durable], func(t *testing.T) {
			e, err := storage.Open(t.TempDir())
			if err != nil {
				t.Fatal(err)
			}
			defer e.Close()
			if err := e.EnsureTable("app", "items", []string{"id"}, nil, nil); err != nil {
				t.Fatal(err)
			}
			if durable {
				tx, err := e.Begin()
				if err != nil {
					t.Fatal(err)
				}
				if err := tx.Insert("app", "items", []string{"durable"}); err != nil {
					t.Fatal(err)
				}
				if err := tx.Commit(); err != nil {
					t.Fatal(err)
				}
			}
			store := &Store{rows: storageRows{engine: e}, definition: formalCatalogDefinition([][]string{{"legacy"}})}
			if err := store.hydrateRowsFromEngine(); err != nil {
				t.Fatal(err)
			}
			rows := store.definition.Namespaces["app"].Tables["items"].Rows
			want := "legacy"
			if durable {
				want = "durable"
			}
			if len(rows) != 1 || rows[0][0] != want {
				t.Fatal("hydration did not select the durable row image", rows, want)
			}
			engineRows, ok := e.SnapshotRows("app", "items")
			if !ok || len(engineRows) != 1 || engineRows[0][0] != want {
				t.Fatal("legacy seed did not reach durable engine", engineRows)
			}
		})
	}
}

func TestFormalCatalogRevision(t *testing.T) {
	calls := 0
	makeRequest := func(revision uint64) *writeRequest {
		return &writeRequest{requireRevision: true, expectedRevision: revision, result: make(chan error, 1), apply: func(d Definition) (Definition, error) { calls++; return d, nil }}
	}
	first, second, stale := makeRequest(7), makeRequest(7), makeRequest(6)
	successful, _ := applyWriteRequests([]*writeRequest{first, second, stale}, formalCatalogDefinition(nil), 7)
	if len(successful) != 1 || calls != 1 {
		t.Fatal("one revision accepted multiple replacements", len(successful), calls)
	}
	for _, request := range []*writeRequest{second, stale} {
		select {
		case err := <-request.result:
			if !errors.Is(err, ErrRevisionConflict) {
				t.Fatal("revision rejection returned wrong error", err)
			}
		default:
			t.Fatal("revision rejection was not returned")
		}
	}
}

func TestFormalOrderedCache(t *testing.T) {
	prior := formalCatalogDefinition([][]string{{"old"}})
	refreshOrderedIndexCaches(Definition{}, &prior)
	old := prior.Namespaces["app"].Tables["items"]
	oldBuilds := 0
	oldBuild := func() ([]OrderedIndexRow, error) { oldBuilds++; return []OrderedIndexRow{{Keys: []string{"old"}}}, nil }
	if _, err := old.CachedOrderedIndex("id", oldBuild); err != nil {
		t.Fatal(err)
	}
	same := cloneDefinition(prior)
	refreshOrderedIndexCaches(prior, &same)
	unchanged := same.Namespaces["app"].Tables["items"]
	if _, err := unchanged.CachedOrderedIndex("id", oldBuild); err != nil {
		t.Fatal(err)
	}
	if oldBuilds != 1 {
		t.Fatal("unchanged image rebuilt a cached index")
	}
	next := formalCatalogDefinition([][]string{{"new"}})
	refreshOrderedIndexCaches(prior, &next)
	current := next.Namespaces["app"].Tables["items"]
	rows, err := current.CachedOrderedIndex("id", func() ([]OrderedIndexRow, error) { return []OrderedIndexRow{{Keys: []string{"new"}}}, nil })
	if err != nil {
		t.Fatal(err)
	}
	if len(rows) != 1 || rows[0].Keys[0] != "new" {
		t.Fatal("changed row image reused a stale index", rows)
	}
	oldRows, err := old.CachedOrderedIndex("id", oldBuild)
	if err != nil {
		t.Fatal(err)
	}
	if oldRows[0].Keys[0] != "old" || oldBuilds != 1 {
		t.Fatal("new cache changed a prior snapshot")
	}
}
