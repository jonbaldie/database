package storage

import (
	"os"
	"strings"
	"testing"
)

func TestFormalSchemaSequence(t *testing.T) {
	dir := t.TempDir()
	e, err := Open(dir)
	if err != nil {
		t.Fatal(err)
	}
	defer e.Close()
	if _, ok := e.SnapshotRows("app", "items"); ok {
		t.Fatal("new engine already has table")
	}
	if err := e.EnsureTable("app", "items", []string{"id"}, []string{"id"}, nil); err != nil {
		t.Fatal(err)
	}
	tx, err := e.Begin()
	if err != nil {
		t.Fatal(err)
	}
	if err := tx.Insert("app", "items", []string{"1"}); err != nil {
		t.Fatal(err)
	}
	if err := tx.Commit(); err != nil {
		t.Fatal(err)
	}
	if err := e.EnsureTable("app", "items", []string{"id", "label"}, []string{"id"}, [][]string{{"label"}}); err != nil {
		t.Fatal(err)
	}
	row, ok := e.LookupPrimary("app", "items", "1")
	if !ok || len(row) != 2 {
		t.Fatal("schema reshape did not rebuild row index", row)
	}
	data, err := os.ReadFile(rowsMetaPath(dir))
	if err != nil || !strings.Contains(string(data), "id,label") {
		t.Fatal("successful schema change was not persisted", string(data), err)
	}
	version := e.tables[tableKey("app", "items")].version
	if err := e.EnsureTable("app", "items", []string{"id", "label"}, []string{"id"}, [][]string{{"label"}}); err != nil {
		t.Fatal(err)
	}
	if e.tables[tableKey("app", "items")].version != version {
		t.Fatal("idempotent schema ensure changed version")
	}
	if err := e.Checkpoint(); err != nil {
		t.Fatal(err)
	}
	if err := e.DropTable("app", "items"); err != nil {
		t.Fatal(err)
	}
	if _, ok := e.SnapshotRows("app", "items"); ok {
		t.Fatal("dropped table stayed visible")
	}
}

func TestFormalSchemaDropRecreate(t *testing.T) {
	dir := t.TempDir()
	e, err := Open(dir)
	if err != nil {
		t.Fatal(err)
	}
	if err := e.EnsureTable("app", "items", []string{"id"}, nil, nil); err != nil {
		t.Fatal(err)
	}
	tx, _ := e.Begin()
	if err := tx.Insert("app", "items", []string{"old"}); err != nil {
		t.Fatal(err)
	}
	if err := tx.Commit(); err != nil {
		t.Fatal(err)
	}
	if err := e.EnsureTable("app", "items", []string{"id", "label"}, nil, nil); err != nil {
		t.Fatal(err)
	}
	if err := e.DropTable("app", "items"); err != nil {
		t.Fatal(err)
	}
	if err := e.EnsureTable("app", "items", []string{"id"}, nil, nil); err != nil {
		t.Fatal(err)
	}
	if err := e.Close(); err != nil {
		t.Fatal(err)
	}
	e, err = Open(dir)
	if err != nil {
		t.Fatal(err)
	}
	defer e.Close()
	rows, ok := e.SnapshotRows("app", "items")
	if !ok || len(rows) != 0 {
		t.Fatal("dropped rows reappeared after table recreation", rows)
	}
}

func TestFormalSchemaPersistenceFailure(t *testing.T) {
	dir := t.TempDir()
	e, err := Open(dir)
	if err != nil {
		t.Fatal(err)
	}
	defer e.Close()
	if err := e.EnsureTable("app", "items", []string{"id"}, nil, nil); err != nil {
		t.Fatal(err)
	}
	if err := os.Mkdir(rowsMetaPath(dir)+".tmp", 0700); err != nil {
		t.Fatal(err)
	}
	if err := e.EnsureTable("app", "items", []string{"id", "label"}, nil, nil); err == nil {
		t.Fatal("schema metadata failure reported success")
	}
	current := e.tables[tableKey("app", "items")]
	if len(current.columns) != 2 {
		t.Fatal("probe did not reach in-memory schema change")
	}
	data, err := os.ReadFile(rowsMetaPath(dir))
	if err != nil || strings.Contains(string(data), "label") {
		t.Fatal("failed metadata write changed durable schema", string(data), err)
	}
}
