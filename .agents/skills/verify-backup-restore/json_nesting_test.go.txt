package main

import (
	"encoding/json"
	"io"
	"strings"
	"testing"
)

func TestFormalJSONNesting(t *testing.T) {
	for _, input := range []string{`0`, `[[0]]`, `{"k":[0]}`, `[]`} {
		decoder := json.NewDecoder(strings.NewReader(input))
		first, err := decoder.Token()
		if err != nil {
			t.Fatal(err)
		}
		if err := consumeBackupJSONValue(decoder, first, 0); err != nil {
			t.Fatalf("valid nested value rejected: %v", err)
		}
		if _, err := decoder.Token(); err != io.EOF {
			t.Fatal("nested JSON value not fully consumed")
		}
	}
	for _, input := range []string{`[0`, `{"k":[0]`} {
		decoder := json.NewDecoder(strings.NewReader(input))
		first, err := decoder.Token()
		if err != nil {
			t.Fatal(err)
		}
		if err := consumeBackupJSONValue(decoder, first, 0); err == nil {
			t.Fatal("truncated JSON accepted")
		}
	}
	for depth := 127; depth <= 130; depth++ {
		input := strings.Repeat("[", depth) + "0" + strings.Repeat("]", depth)
		decoder := json.NewDecoder(strings.NewReader(input))
		first, err := decoder.Token()
		if err != nil {
			t.Fatal(err)
		}
		err = consumeBackupJSONValue(decoder, first, 0)
		if depth <= 128 && err != nil {
			t.Fatal("valid depth rejected")
		}
		if depth > 128 && err == nil {
			t.Fatal("excess JSON depth accepted")
		}
	}
	if err := validateBackupCatalogReader(strings.NewReader(`{"namespaces":{}} {}`)); err == nil {
		t.Fatal("trailing JSON accepted")
	}
}
