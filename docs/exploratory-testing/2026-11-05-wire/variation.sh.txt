#!/bin/bash
set -euo pipefail
cd "$(dirname "$0")/../../.."
root="$PWD"; evidence="$root/docs/exploratory-testing/2026-11-05-wire"; scratch="$root/.explore-20261105"
export GOMODCACHE="$scratch/mod" GOCACHE="$scratch/cache" TMPDIR="$scratch/tmp" GOTOOLCHAIN=auto GOMAXPROCS=2 GOFLAGS='-p=2 -mod=readonly'
pid=''
cleanup() { if [[ -n "$pid" ]]; then kill -TERM "$pid" 2>/dev/null || :; wait "$pid" 2>/dev/null || :; echo "cleanup stopped $pid"; fi; }
trap cleanup EXIT
trap 'exit 130' INT
trap 'exit 143' TERM
verify_command() ( label="$1"; shift; dir="$(mktemp -d "$evidence/$label.XXXXXX")"; printf '%s\n' "$@" > "$dir/command.txt"; set +e; "$@" > "$dir/output.log" 2>&1; status=$?; echo "$status" > "$dir/exit-status.txt"; echo "$label exit=$status evidence=$dir"; exit "$status" )
set +e
verify_command restore-nonempty "$root/bin/database" restore --backup "$scratch/backup.tar" --data-directory "$scratch/data" --result=json
status=$?
set -e
if [[ "$status" != 3 ]]; then echo "expected precondition exit 3, got $status"; exit 1; fi
"$root/bin/database" serve --data-directory "$scratch/data" --mysql-listen-address=127.0.0.1:43459 --diagnostics-listen-address=127.0.0.1:43460 > "$evidence/server-variation.log" 2>&1 & pid=$!
echo "registered pid=$pid mysql=43459 diagnostics=43460"
for i in $(seq 1 100); do if curl -fsS http://127.0.0.1:43460/ready >/dev/null 2>&1; then break; fi; kill -0 "$pid"; sleep .2; done
verify_command client-after-rejected-restore go run "$scratch/client.go" reopen
kill -TERM "$pid"; wait "$pid"; echo "stopped pid=$pid"; pid=''
